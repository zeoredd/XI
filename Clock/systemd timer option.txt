Systemd timers (alternative to cron)
Prep (one-time)
sudo mkdir -p /home/node-alpha/XI/logs
sudo chown -R node-alpha:node-alpha /home/node-alpha/XI/logs

1) 03:00 — Protocol audit + AI edge-cases task

/etc/systemd/system/xi-sentinel-night-audit.service

[Unit]
Description=XI Sentinel Night Audit (protocol + AI edge-cases)

[Service]
Type=oneshot
WorkingDirectory=/home/node-alpha/XI
Environment=PYTHONUNBUFFERED=1
ExecStart=/home/node-alpha/XI/venv/bin/python3 -m runtime.routines_orchestrator --sentinel --sentinel-ai-edgecases
StandardOutput=append:/home/node-alpha/XI/logs/sentinel_night_audit.log
StandardError=append:/home/node-alpha/XI/logs/sentinel_night_audit.log
User=node-alpha
Group=node-alpha


/etc/systemd/system/xi-sentinel-night-audit.timer

[Unit]
Description=Run XI Sentinel Night Audit at 03:00 daily

[Timer]
OnCalendar=*-*-* 03:00:00
Persistent=true
AccuracySec=1min
Unit=xi-sentinel-night-audit.service

[Install]
WantedBy=timers.target

Monthly reindex/backup (1st 05:30)

/etc/systemd/system/xi-monthly-index-backup.service

[Unit]
Description=XI Monthly Cold Index Sweep + DB Backup

[Service]
Type=oneshot
WorkingDirectory=/home/node-alpha/XI
Environment=PYTHONUNBUFFERED=1
ExecStart=/bin/sh -c '/home/node-alpha/XI/venv/bin/python3 -m abilities.common_abilities.memory_indexer >> /home/node-alpha/XI/logs/index_full.log 2>&1 && /usr/bin/pg_dump xi_memory > /home/node-alpha/XI/backups/xi_memory_$(date +%Y%m%d).sql'
StandardOutput=append:/home/node-alpha/XI/logs/index_full.log
StandardError=append:/home/node-alpha/XI/logs/index_full.log
User=node-alpha
Group=node-alpha


/etc/systemd/system/xi-monthly-index-backup.timer

[Unit]
Description=Run XI Monthly Index Sweep + Backup (1st @ 05:30)

[Timer]
OnCalendar=*-01 05:30:00
Persistent=true
AccuracySec=1min
Unit=xi-monthly-index-backup.service

[Install]
WantedBy=timers.target

Enable everything
sudo systemctl daemon-reload
sudo systemctl enable --now \
  xi-sentinel-night-audit.timer \
  xi-sentinel-ai-consume.timer \
  xi-run-all.timer \
  xi-sentinel-quick.timer \
  xi-sandbox-retention.timer \
  xi-db-vacuum.timer \
  xi-monthly-index-backup.timer
